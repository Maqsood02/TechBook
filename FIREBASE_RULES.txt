rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helper Functions ──
    function signedIn() {
      return request.auth != null;
    }

    function hasEmail() {
      return signedIn() && request.auth.token != null && ('email' in request.auth.token) && request.auth.token.email != null;
    }

    function userEmail() {
      return hasEmail() ? request.auth.token.email.lower() : '';
    }

    function getUsername() {
      return !hasEmail() ? '' :
        (userEmail().split('@')[0] == 'cof.techbook' ? 'cof@techbook' : userEmail().split('@')[0]);
    }

    function isAdmin() {
      return signedIn() && hasEmail() && (
        exists(/databases/$(database)/documents/admins/$(getUsername())) ||
        exists(/databases/$(database)/documents/admins/$(userEmail())) ||
        userEmail() == 'techbook.com@techbook.ac.in' ||
        userEmail() == 'cof.techbook@techbook.ac.in' ||
        userEmail() == 'techbook.ac.in@gmail.com'
      );
    }

    function isOwner(studentId) {
      return signedIn() && hasEmail() && (
        userEmail() == studentId.lower() + '@techbook.ac.in' ||
        userEmail() == studentId.lower()
      );
    }

    // ── Students Collection ──
    match /students/{studentId} {
      allow read, write: if isAdmin() || isOwner(studentId);
      allow create: if signedIn() && !exists(/databases/$(database)/documents/admins/$(studentId.lower())) && isOwner(studentId);
    }

    // ── Attendance Collection ──
    match /attendance/{attendanceId} {
      allow read: if isAdmin() || (signedIn() && hasEmail() && (
        resource == null ||
        attendanceId.split('_')[0].lower() + '@techbook.ac.in' == userEmail() ||
        resource.data.usn.lower() == userEmail().split('@')[0] ||
        resource.data.usn.lower() + '@techbook.ac.in' == userEmail()
      ));
      allow create: if signedIn() && (
        isAdmin() || (hasEmail() && (
          request.resource.data.usn.lower() + '@techbook.ac.in' == userEmail() ||
          attendanceId.split('_')[0].lower() + '@techbook.ac.in' == userEmail()
        ))
      );
      allow update, delete: if isAdmin() || (signedIn() && hasEmail() && (
        attendanceId.split('_')[0].lower() + '@techbook.ac.in' == userEmail() ||
        (resource != null && resource.data.usn.lower() + '@techbook.ac.in' == userEmail())
      ));
    }

    // ── Attendance Codes ──
    match /attendance_codes/{codeId} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    // ── Password Reset Requests ──
    match /password_reset_requests/{id} {
      allow create: if request.resource.data.keys().hasAll(['usn', 'timestamp']) &&
                       request.resource.data.usn is string &&
                       request.resource.data.usn.size() > 0 &&
                       request.resource.data.usn.size() <= 20;
      allow read, update, delete: if isAdmin();
    }

    // ── Settings ──
    match /settings/{settingId} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    // ── Quizzes & Results ──
    match /quizzes/{id} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    match /quiz_results/{id} {
      allow read: if isAdmin() || (signedIn() && hasEmail() && (
        resource == null ||
        id.split('_')[0].lower() + '@techbook.ac.in' == userEmail() ||
        resource.data.usn.lower() == userEmail().split('@')[0] ||
        resource.data.usn.lower() + '@techbook.ac.in' == userEmail()
      ));
      allow create: if signedIn() && (
        isAdmin() || (hasEmail() && (
          request.resource.data.usn.lower() + '@techbook.ac.in' == userEmail() ||
          id.split('_')[0].lower() + '@techbook.ac.in' == userEmail()
        ))
      );
      allow update, delete: if isAdmin();
    }

    // ── Lecture Notes & Chunks ──
    match /techbook_notes/{id} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }
    match /techbook_notes/{id}/chunks/{chunkId} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    // ── Question Bank & Chunks ──
    match /qbank_papers/{id} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }
    match /qbank_papers/{id}/chunks/{chunkId} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    // ── PYQ Papers & Chunks ──
    match /pyq_papers/{id} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }
    match /pyq_papers/{id}/chunks/{chunkId} {
      allow read: if signedIn();
      allow write: if isAdmin();
    }

    // ── Help Threads & Messages ──
    match /help_threads/{tid} {
      allow read, write: if isAdmin() || (signedIn() && hasEmail() && tid.split('_')[0].lower() + '@techbook.ac.in' == userEmail());
    }
    match /help_messages/{tid}/msgs/{msgId} {
      allow read, write: if isAdmin() || (signedIn() && hasEmail() && tid.split('_')[0].lower() + '@techbook.ac.in' == userEmail());
    }

    // ── Admin Profiles & UIDs ──
    match /admins/{adminId} {
      allow read, write: if isAdmin() || (signedIn() && hasEmail() && userEmail() == adminId.lower() + '@techbook.ac.in');
    }

    match /admins_uids/{uid} {
      allow read, write: if isAdmin();
    }

    // ── Promos (Public view, Admin manage) ──
    match /promos/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // ── User Sessions & Tracking ──
    match /user_sessions/{sessionId} {
      allow read: if isAdmin();
      allow write: if isAdmin() || (signedIn() && (request.auth.uid == sessionId || (hasEmail() && sessionId.lower() == userEmail().split('@')[0])));
    }

    match /user_activities/{activityId} {
      allow read: if isAdmin();
      allow write: if signedIn();
    }

    // ── Founder Messages ──
    match /founder_messages/{id} {
      allow read, write: if isAdmin();
    }

    // ── Launches ──
    match /launches/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
  }
}
